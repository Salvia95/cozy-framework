# 실패하고 cozy sync behaviors가 정리해 다시 만든다. 다른 모듈의 컴포넌트를 쓰면 R16이 실패한다. 레이아웃이 모듈 파일을 싣고 mvn verify가
# 화면을 그리고 만든 JS의 문법을 읽는다(C10)
cozy add crud task --entity Task --label "할 일" --icon list-details --field title:text!:제목
cozy add module note --label 노트
cozy add behavior task title-count --purpose "제목의 글자 수를 보인다" --state title:text@data-title --computed length:length(title) --computed "label:format('제목 {length}자')"
cozy add behavior note picker --purpose "누를 때마다 고름과 풂을 바꾼다" --state picked:bool --action flip:toggle(picked)
cozy check
# 명세를 손으로 다시 썼다(내용은 같다): 다시 만들기 전에는 C12, cozy sync behaviors가 정리한다
